       const providers = persistedState.providers.map((p: ProviderConfig) 
     => {
         // Fix OpenRouter baseUrl (was wrong domain)
         if (p.name?.toLowerCase().includes('openrouter') &&
             !p.baseUrl?.includes('openrouter.ai')) {
           p.baseUrl = 'https://openrouter.ai/api/v1';
         }
         // Fix Z.AI model case sensitivity
         if (p.baseUrl?.includes('api.z.ai') && p.model) {
           p.model = p.model.toLowerCase();
         }
         return p;
       });

       return { ...persistedState, providers };
     };

     // In persist config:
     {

       version: PROVIDER_STORE_VERSION,
